        lng operator()(lng a, lng b) const {

        }
    };

        lng operator()(lng a, lng b) const {

        }
    };

2.2.3 Square Root Decomposition

// O(1)
auto add = [&](int i) {
    if (freq[a[i]] == 0) {
        cur_ans++;
    }
    freq[a[i]]++; 
};

// O(1)
auto del = [&](int i) {
    freq[a[i]]--; 
    if (freq[a[i]] == 0) {
        cur_ans--;
    }
};

auto calc = [&](int q_idx) {
    ans[q_idx] = cur_ans;
};
